Private Process ReceiveProcess () 
Var receiveddata As ByteString [255]
Var MessageBody As ByteString [255]
	Wait(0:1.0)
	If DebugMessage Then DebugPrintFormat("{t} {t} ReceiveProcess started", DeviceName, IPAddress) End If
	Do While True
		If TcpClientConnector.WaitForData(365:0:0:0) Then // timeout 1 year or ~ infinity
//		If	TcpClientConnector.ReceiveAnything(receiveddata, 0:5.0) Then
			TcpClientConnector.ReceiveAnything(receiveddata, 0:1.0)
			TcpClientConnector.ClearReceiveBuffer()
//			ReceivedDataProcess(receiveddata)
//		If TcpClientConnector.WaitForData(0:5:0) Then // timeout 1 year or ~ infinity
//		If TcpClientConnector.Receive(receiveddata, 0:5.0) Then 
//			TcpClientConnector.Receive(receiveddata)
//			TcpClientConnector.ClearReceiveBuffer()
			
			If DebugMessage Then DebugPrintFormat("{t} {t} RxBuffer >> {s}", DeviceName, IPAddress, receiveddata) End If
			If GetMessageBodyFromBuffer (receiveddata, MessageBody) = True Then
				If DebugMessage Then DebugPrintFormat("{t} {t} MessageBody >> {s}", DeviceName, IPAddress,MessageBody) End If
				ProcessMessageBody (MessageBody)
			End If
		Else
			If DebugMessage Then DebugPrintFormat("{t} {t} Connector.WaitForData - FAILED with error {n}, try to recconect", DeviceName, IPAddress, GetLastError()) End If
			Wait(0:0.3)
//			If ConnectionAttemptCounter <3 Then Connect() End If
			Connect()
		End If
	End Do
End Process


Private Function GetMessageBodyFromBuffer (ByRef buffer As ByteString, ByRef messagebody As ByteString) As Boolean
Private Var StartPosition As Long 
Private Var EndPosition As Long 
Private Var ActualLength As Long
Private Var AnswerValidity As ByteString [1]
Private Var buffer1st As ByteString [50]
	
	// clear strange \00 value from beginning
	GetByteStringLeft(buffer,1,buffer1st)
	If CompareByteString(buffer1st, &s"\00") = 2 Then
		TrimByteStringLeft(buffer,buffer,&S"\00")
	End If

	StartPosition := GetSubByteStringPosition (buffer, MESSAGE_HEADER)
	If StartPosition > 0 Then
		// Answer validity check - ACK or NAK
		GetByteStringMiddle(buffer,5,1,AnswerValidity)
		If AnswerValidity=&s"A" Then // answer is correct, proceed
			GetByteStringMiddle(buffer,4,1,DataLenghtStr)
//			debugprint(DataLenghtStr)
			DataLenght:=DataLenghtStr[1]
//			debugprint(DataLenght)
			EndPosition := 4+DataLenght
		 	GetByteStringMiddle (buffer, 6, EndPosition - 5, messagebody)// Extract message body from buffer, incl. CMD code as first byte
//			debugprint(messagebody)		
//					ActualLength := GetByteStringActualLength(buffer)				    
//					GetByteStringRight (buffer, ActualLength - EndPosition - MessageFooterLenght + 1, buffer)
			
			Return Value True		
		Else // answer error
//			DebugPrint("ERROR - Feedback value is not valid - no ACK")
//			Controller1.Files_Log.WriteLogFile_Alerts("Feedback value is not valid - no ACK")
			Return Value False
		End If
		
	Else
		DebugPrint("ERROR - Feedback value do not have Header")
		Return Value False	
	End If	
End Function


Private Function ProcessMessageBody (messagebody As ByteString)

//	AnswerPos:=GetSubByteStringPosition(messagebody, &s"=")
//	MessageLen:=GetByteStringActualLength(messagebody)	
	GetByteStringLeft(messagebody,1, AnswerType)
//	debugprint(&S"AnswerPos",AnswerPos)
//	debugprint(&S"AnswerType", AnswerType)
	Select AnswerType
	
		Case &s"\11"// POWER STATUS
			GetByteStringMiddle(messagebody, 2, 1, PowerStatusStr)
			If PowerStatusStr<>PowerStatusStrPrev Then
				PowerStatusStrPrev:=PowerStatusStr
				Select PowerStatusStr
					Case &s"\00"
						PowerStatus:=0
						RaiseEvent OnPowerStatusChange (PowerStatus)
					Case &s"\01"
						PowerStatus:=1
						RaiseEvent OnPowerStatusChange (PowerStatus)
					Case Else
						PowerStatus:=-1
						RaiseEvent OnPowerStatusChange (PowerStatus)
				End Select
//				debugprint(PowerStatus)
			End If
			

		Case &s"\0B" //Serial number
			GetByteStringMiddle(messagebody, 2, 14, SerialNumberStr)
//			ByteStringToText(SerialNumberStr,SerialNumber,_ASCII_TO_TEXT)
//			debugprint("SerialNumber>>>", SerialNumber)	
			RaiseEvent OnDeviceInfo_SerialNumber (SerialNumberStr)
			
		Case &s"\0D" //DEVICE STATUS
//			GetByteStringMiddle(messagebody, AnswerPos+1, MessageLen-AnswerPos, Lamp1HoursStr)
//			If Lamp1HoursStr<>Lamp1HoursStrPrev Then
//				Lamp1HoursStrPrev:=Lamp1HoursStr
//				ParseByteString(Lamp1HoursStr,&S"{n}", Lamp1Hours)
//				RaiseEvent OnLamp1HoursChange (Lamp1Hours)
//			End If
			Temperature:=messagebody[6]
			If Temperature<>TemperaturePrev Then
				TemperaturePrev:=Temperature
				RaiseEvent OnTemperatureChange (Temperature)
			End If
//Global Event OnError_Fan1 As Process (ErrorCode As ByteString)
//Global Event OnError_Lamp1 As Process (ErrorCode As ByteString)
//Global Event OnError_Temp1 As Process (ErrorCode As ByteString)

		Case &s"\0E" //Software version
			GetByteStringMiddle(messagebody, 2, 18, SoftVersionStr)
//			DebugPrintFormat("SoftVersion>>> {s}", SoftVersionStr)	
			RaiseEvent OnDeviceInfo_Firmware (SoftVersionStr)
//			ByteStringToText(SoftVersionStr,SoftVersion,_ASCII_TO_TEXT)
//			RaiseEvent OnDeviceInfo_SoftwareVersion (SoftVersion)

		Case &s"\8A" //model name
			GetByteStringMiddle(messagebody, 2, 18, ModelStr)	
			RaiseEvent OnDeviceInfo_Model (ModelStr)
						
		Case Else
			DebugPrint(&s"OnMessage",messagebody)
			RaiseEvent OnMessage (messagebody)
	End Select

End Function

Public Function Connect () As Boolean
Private Var errorrecnumber As Long
Private Var i As Long
Private Var TxtStr As ByteString [50]
Private Var LanConnectionIsOK As Boolean
Var IPAddressInfo As Text [15]
Var port As Long
Var ErrorNum As Long
	For i:=1 To 3
		If Not TcpClientConnector.Connect (IPAddress, TCPPort, Timeout) Then
	//		RaiseEvent OnError_TelnetNetwork (GetLastError())
			ErrorNum:=GetLastError()
			If DebugMessage Then DebugPrintFormat("{t} {t} LAN connection FAILED with error {n}", DeviceName, IPAddress, ErrorNum) End If
//			DebugPrint (TxtStr)
			If ErrorNum=89 Then // channel already connected
				LanConnectionIsOK:=True
				Exit For
			Else // other error		
				LanConnectionIsOK:=False
			End If
		Else
			If DebugMessage Then DebugPrintFormat ("{t} {t} connected and LAN connection OK", DeviceName, IPAddress) End If
			LanConnectionIsOK:=True
			If  CommunicationErrorIsRaised=True Then 
				CommunicationErrorIsRaised:=False
//				RaiseEvent OnRecovery (&s"LAN Recovery")
				RaiseEvent OnRecovery_Communication()
			End If 
			Exit For
		End If
	End For
	If LanConnectionIsOK=False And CommunicationErrorIsRaised=False Then 
		CommunicationErrorIsRaised:=True 
		RaiseEvent OnError_Communication (&s"LAN error") //ErrorNum ?
		Return Value False
	Else
		Return Value True
	End If
End Function

Global Function PowerOn_WOL_MAC_WithStatusCheck (MAC As Text) As Boolean
Var i As Long
Var LanConnectionIsOK As Boolean
Var ErrorNum As Long
	WakeOnLAN(MAC)
	For i:=1 To 30
		If Not TcpClientConnector.Connect (IPAddress, TCPPort, 0:11.0) Then
			ErrorNum:=GetLastError()
			DebugPrintFormat("{t} {t} after WOL - FAILED {n} with error {n}", DeviceName, IPAddress, i, ErrorNum)
			If ErrorNum=89 Then // channel already connected
				LanConnectionIsOK:=True
				Exit For
			Else // other error		
				LanConnectionIsOK:=False
				WakeOnLAN(MAC)
			End If
		Else
			DebugPrintFormat ("{t} {t} after WOL - connection OK", DeviceName, IPAddress)
			LanConnectionIsOK:=True
			RaiseEvent OnRecovery_Communication()
			Exit For
		End If
	End For	
	Return Value LanConnectionIsOK
End Function

Global Function DisplayStatusCheckLoop_Enable()
	StartProcess ReceiveProcess ()
	StartProcess StatusRequest_FastLoop ()
	StartProcess StatusRequest_SlowLoop ()
End Function


Global Function DisplayStatusCheckLoop_Disable()
	StopProcess ReceiveProcess ()
	StopProcess StatusRequest_FastLoop ()
	StopProcess StatusRequest_SlowLoop () 
End Function


/// Overwrite CVC configuration value ///

Global Function OverwriteCVCconfig_Device_Model(ModelN As Byte) // Use correct driver type
	ModelName:=ModelN
End Function

Global Function OverwriteCVCconfig_Device_IPaddress(IPAddressNew As Text)
	DisplayStatusCheckLoop_Disable()
	IPAddress:=IPAddressNew
	Wait(0:1.0)
	DisplayStatusCheckLoop_Enable()
End Function

	TCPPort:=TCPPortNew
End Function

	MacAddress:=MacAddressNew
End Function
